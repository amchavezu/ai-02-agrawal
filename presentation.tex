\begin{frame}{1 \textperiodcentered{} The paper and the agent's problem}
  \textbf{Question.} How do cognitive tools change effort, productivity, and the
  value of implementation skill and judgment?

  \medskip
  \textbf{Single mechanism.} A tool substitutes for implementation effort but can
  complement opportunity and payoff judgment.

  \medskip
  Conditional on finding an opportunity, the agent chooses $e_t\geq0$:
  \[
    \max_{e_t\geq0}\;M(e_t;\theta)
    =p(se_t;\theta)\alpha\Delta-c(e_t;\theta).
  \]

  \begin{itemize}\small
    \item $s$: implementation skill; $\alpha$: payoff judgment.
    \item $\gamma(t)$: opportunity judgment; $\theta$: tool quality.
    \item Interior FOC: $p'(se_t^*;\theta)s\alpha\Delta=c'(e_t^*;\theta)$.
  \end{itemize}
\end{frame}

\begin{frame}{2 \textperiodcentered{} Proposition 1, with its conditions}
  \textbf{Cognitive-tool conditions.} For $\theta'>\theta$ and every $e$,
  \[
    p(se;\theta')\geq p(se;\theta),\qquad
    c(e;\theta')\leq c(e;\theta),
  \]
  while $p'(se;\theta)/c'(e;\theta)$ strictly decreases in $\theta$ for $e>0$.

  \medskip
  With increasing/concave $p$, increasing/convex $c$, and an interior optimum:
  \[
    e_t^*(1)<e_t^*(0),\qquad e_t^*(\theta)=e^*(\theta),
    \qquad V_0(\theta)>V_0(0).
  \]

  \footnotesize\emph{Intuition.} Tool adoption raises achievable net benefit but
  lowers the marginal return to direct human implementation effort.
\end{frame}

\begin{frame}{3 \textperiodcentered{} What I did}
  \begin{itemize}
    \item Rewrote continuation value as $V(\theta)=H+\theta Z$, where
      $H=(\Delta^2/4)\Gamma\alpha^2s$ and $Z=\Gamma/s$.
    \item Derived
      $\operatorname{Var}(V)=\operatorname{Var}(H)
      +2\theta\operatorname{Cov}(H,Z)+\theta^2\operatorname{Var}(Z)$.
    \item Relaxed $\Gamma\perp s$: initial equalisation now requires
      $\mathbb E[\Gamma^2]<\mathbb E[\Gamma s]\mathbb E[\Gamma/s]$.
    \item Wrote a standard-library Python check with three distributions and
      reproducible CSV, text, and SVG outputs.
  \end{itemize}

  \medskip
  \textbf{Counterexample output:} condition (30) holds, but
  $\theta^*=1.792$ and
  $\left.d\operatorname{Var}(V)/d\theta\right|_{\theta=1}=-0.0727$.
\end{frame}

\begin{frame}{4 \textperiodcentered{} Where I did not believe the AI}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.64\textwidth}
      \textbf{Loose claim.} ``Variance is U-shaped in AI quality.''

      \medskip
      \textbf{Audit.} The object is $\operatorname{Var}(V(\theta))$.
      Condition (30) makes the initial slope negative and implies
      $\theta^*>0$, but a positive slope at $\theta=1$ also requires
      \[
        \boxed{\theta^*<1}.
      \]
      The paper's Appendix A.3 proves a positive slope of the derivative, not
      that its zero occurs before one.

      \medskip
      \textbf{Verdict.} The AI's unconditional statement is incorrect, and the
      paper's displayed endpoint claim is under-conditioned.
