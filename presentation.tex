\begin{frame}{1. A tool changes implementation, not the objective}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.49\textwidth}
      {\color{Teal}\textbf{Question}}\\[-1pt]
      How do cognitive tools change effort, productivity, and the value of human skills?

      \medskip
      {\color{Teal}\textbf{Single mechanism}}\\[-1pt]
      The tool directly improves implementation, lowering the marginal return to human
      implementation effort; judgment still selects payoffs and opportunities.

      \medskip
      \footnotesize
      \textbf{Skills:} implementation \(s\), payoff judgment \(\alpha\), and opportunity
      judgment \(\{\gamma(t)\}\).
    \end{column}
    \begin{column}{0.47\textwidth}
      {\color{Teal}\textbf{Conditional agent problem}}
      \[
        \max_{e_t\geq0}\;
        \underbrace{p(se_t;\theta)\alpha\Delta-c(e_t;\theta)}_{M(e_t;\theta)}.
      \]
      The choice variable is effort \(e_t\); the constraint is \(e_t\geq0\).

      \medskip
      At an interior optimum,
      \[
        p'(se_t^*;\theta)s\alpha\Delta=c'(e_t^*;\theta).
      \]
      \footnotesize Opportunity judgment changes how often the problem occurs, not this
      conditional FOC.
    \end{column}
  \end{columns}
  \vfill
  {\color{Muted}\scriptsize Source: Agrawal, Gans \& Goldfarb (2025), Sections 2--3.}
\end{frame}

\begin{frame}{2. Proposition 1: more value with less effort}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\textwidth}
      {\color{Teal}\textbf{All conditions used for strict conclusions}}
      \begin{itemize}\small
        \item \(p\) increasing and weakly concave; \(c\) increasing and weakly convex.
        \item \(s>0\), \(\alpha\Delta>0\), and a finite positive opportunity multiplier
          \(\Gamma\).
        \item Interior, unique optima; at least one relevant level improvement is strict.
        \item For every feasible \(e\):
      \end{itemize}
      \vspace{-0.35cm}
      \[
        p(se;1)\ge p(se;0),\qquad c(e;1)\le c(e;0),
      \]
      \vspace{-0.35cm}
      \[
        \frac{p'(se;1)}{c'(e;1)}<
        \frac{p'(se;0)}{c'(e;0)}\quad(e>0).
      \]
    \end{column}
    \begin{column}{0.48\textwidth}
      {\color{Teal}\textbf{Result}}
      \[
        \boxed{e_t^*(1)<e_t^*(0)}
      \]
      \[
        \boxed{e_t^*(\theta)=e^*(\theta)\quad\forall t}
      \]
      \[
        \boxed{V_0(1)>V_0(0)}
      \]
      \keybox{\small The tool raises attainable net value but lowers the marginal payoff
      to direct implementation effort. Conditional choices are time-invariant because the
      one-opportunity objective is time-invariant.}
    \end{column}
  \end{columns}
  \vfill
  {\color{Muted}\scriptsize At a corner or without strictness, the safe comparative statics
  are weak. Source: Proposition 1 and Appendix A.1.}
\end{frame}

\begin{frame}{3. Dependence, not marginals, governs the variance effect}
  \vspace{-0.25cm}
  \[
    V(\theta)=H+\theta Z,\quad
    H=\frac{\Delta^2}{4}\Gamma\alpha^2s,\quad Z=\frac{\Gamma}{s},
  \]
  \vspace{-0.25cm}
  \[
    \operatorname{Var}(V)=\operatorname{Var}(H)
    +2\theta\operatorname{Cov}(H,Z)+\theta^2\operatorname{Var}(Z),
  \]
  \vspace{-0.20cm}
  \[
    \left.\frac{d\operatorname{Var}(V)}{d\theta}\right|_0<0
    \iff
    \mathbb E[\Gamma^2]<\mathbb E[\Gamma s]\mathbb E[\Gamma/s].
  \]

  \vspace{-0.10cm}

\begin{frame}{4. I did not believe the endpoint claim}
  \vspace{-0.06cm}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.54\textwidth}
      {\color{Teal}\textbf{Loose AI answer:}}\quad
      \emph{Variance is U-shaped in AI quality.}

      \smallskip
      {\color{Teal}\textbf{Audit}}
      \[
        \begin{aligned}
        \operatorname{Var}'(V;\theta)
          &=2\operatorname{Cov}(H,Z)+2\theta\operatorname{Var}(Z),\\
        \operatorname{Var}'(V;1)>0
          &\iff \boxed{\theta^*<1}.
        \end{aligned}
      \]
      Condition (30) proves \(\operatorname{Var}'(V;0)<0\) and \(\theta^*>0\),
      but not the endpoint inequality.

      \smallskip
      \textbf{Counterexample:} (30) holds, but \(\theta^*=1.792\) and
      the slope at one is \(-0.0727\).

      \smallskip
      \keybox{\small\textbf{Verdict:} the unconditional AI statement is incorrect;
      the paper's endpoint claim is under-conditioned. The object is
      \(\operatorname{Var}_i(V_i(\theta))\).}
