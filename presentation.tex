\begin{frame}{1 \textperiodcentered{} The paper and the agent's problem}
  \textbf{Question.} How do cognitive tools change effort, productivity, and the
  value of implementation skill and judgment?

  \medskip
  \textbf{Single mechanism.} A tool substitutes for implementation effort but can
  complement opportunity and payoff judgment.

  \medskip
  Conditional on finding an opportunity, the agent chooses $e_t\geq0$:
  \[
    \max_{e_t\geq0}\;M(e_t;\theta)
    =p(se_t;\theta)\alpha\Delta-c(e_t;\theta).
  \]

  \begin{itemize}\small
    \item $s$: implementation skill; $\alpha$: payoff judgment.
    \item $\gamma(t)$: opportunity judgment; $\theta$: tool quality.
    \item Interior FOC: $p'(se_t^*;\theta)s\alpha\Delta=c'(e_t^*;\theta)$.
  \end{itemize}
\end{frame}

\begin{frame}{2 \textperiodcentered{} Proposition 1, with its conditions}
  \textbf{Cognitive-tool conditions.} For $\theta'>\theta$ and every $e$,
  \[
    p(se;\theta')\geq p(se;\theta),\qquad
    c(e;\theta')\leq c(e;\theta),
  \]
  while $p'(se;\theta)/c'(e;\theta)$ strictly decreases in $\theta$ for $e>0$.

  \medskip
  With increasing/concave $p$, increasing/convex $c$, and an interior optimum:
  \[
    e_t^*(1)<e_t^*(0),\qquad e_t^*(\theta)=e^*(\theta),
    \qquad V_0(\theta)>V_0(0).
  \]

  \footnotesize\emph{Intuition.} Tool adoption raises achievable net benefit but
  lowers the marginal return to direct human implementation effort.
\end{frame}
